cmake_minimum_required(VERSION 3.20)
project(AegisWire LANGUAGES C)
set(CMAKE_C_STANDARD 17)
if(NOT WIN32)
 message(FATAL_ERROR "This project requires Windows")
endif()
set(NPCAP_SDK "" CACHE PATH "Npcap SDK directory")
if(NOT EXISTS "${NPCAP_SDK}/Include/pcap.h")
 message(FATAL_ERROR "Pass -DNPCAP_SDK=C:/Npcap-SDK")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
 set(PCAP_LIB "${NPCAP_SDK}/Lib/x64")
else()
 set(PCAP_LIB "${NPCAP_SDK}/Lib")
endif()
add_executable(AegisWire WIN32 src/main.c)
target_include_directories(AegisWire PRIVATE "${NPCAP_SDK}/Include")
target_link_directories(AegisWire PRIVATE "${PCAP_LIB}")
target_link_libraries(AegisWire PRIVATE wpcap Packet ws2_32)

if(MSVC)
  target_compile_options(AegisWire PRIVATE /W4)
endif()
